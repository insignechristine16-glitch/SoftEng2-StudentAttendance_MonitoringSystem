PRAGMA foreign_keys = ON;

CREATE TABLE users (
  id            INTEGER PRIMARY KEY,
  email         TEXT NOT NULL UNIQUE COLLATE NOCASE,
  password_hash TEXT NOT NULL,
  name          TEXT NOT NULL,
  role          TEXT NOT NULL CHECK (role IN ('student','instructor','administrator'))
);

CREATE TABLE sections (
  id            INTEGER PRIMARY KEY,
  code          TEXT NOT NULL UNIQUE,
  subject       TEXT NOT NULL,
  room          TEXT NOT NULL,
  schedule      TEXT NOT NULL,
  instructor_id INTEGER NOT NULL REFERENCES users(id)
);

CREATE TABLE students (
  id             INTEGER PRIMARY KEY,
  student_number TEXT NOT NULL UNIQUE,
  name           TEXT NOT NULL,
  section_id     INTEGER NOT NULL REFERENCES sections(id),
  active         INTEGER NOT NULL DEFAULT 1 CHECK (active IN (0,1)),  -- soft delete
  user_id        INTEGER NOT NULL UNIQUE REFERENCES users(id)         -- email lives in users
);

CREATE TABLE class_sessions (
  id           INTEGER PRIMARY KEY,
  section_id   INTEGER NOT NULL REFERENCES sections(id),
  session_date TEXT NOT NULL,                                          -- YYYY-MM-DD
  created_by   INTEGER NOT NULL REFERENCES users(id),
  created_at   TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE (section_id, session_date)                                    -- no duplicate sessions
);

CREATE TABLE attendance (
  id         INTEGER PRIMARY KEY,
  session_id INTEGER NOT NULL REFERENCES class_sessions(id) ON DELETE CASCADE,
  student_id INTEGER NOT NULL REFERENCES students(id),
  status     TEXT NOT NULL CHECK (status IN ('present','late','absent','excused')),
  UNIQUE (session_id, student_id)
);

CREATE TABLE edit_log (
  id         INTEGER PRIMARY KEY,
  session_id INTEGER NOT NULL REFERENCES class_sessions(id) ON DELETE CASCADE,
  user_id    INTEGER NOT NULL REFERENCES users(id),
  reason     TEXT NOT NULL,
  edited_at  TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_students_section ON students(section_id);
CREATE INDEX idx_attendance_session ON attendance(session_id);
CREATE INDEX idx_attendance_student ON attendance(student_id);

import sqlite3
from pathlib import Path
from flask import current_app, g

SCHEMA = Path(__file__).with_name('schema.sql')


def connect(path):
    con = sqlite3.connect(path)
    con.row_factory = sqlite3.Row
    con.execute('PRAGMA foreign_keys = ON')
    return con


def get_db():
    if 'db' not in g:
        g.db = connect(current_app.config['DATABASE'])
    return g.db


def close_db(_=None):
    con = g.pop('db', None)
    if con:
        con.close()


def init_app(app):
    app.teardown_appcontext(close_db)


def init_schema(path):
    """Create a fresh database file from schema.sql (used by seed.py)."""
    con = connect(path)
    con.executescript(SCHEMA.read_text())
    return con

#validators.py
import re
from datetime import datetime

ROLES = ('student', 'instructor', 'administrator')
STATUSES = ('present', 'late', 'absent', 'excused')
EMAIL = re.compile(r'[^@\s]+@[^@\s]+\.[^@\s]+')
STUDENT_ID = re.compile(r'\d{4}-\d{4}')


def validate_login(form):
    """Returns (clean_data, [error messages])."""
    role = form.get('role', '').strip()
    ident = form.get('identifier', '').strip()
    pw = form.get('password', '')
    errors = []
    if role not in ROLES:
        errors.append('Choose a valid role.')
    if not ident:
        errors.append('Enter your student ID or school email.')
    elif len(ident) > 120:
        errors.append('That ID or email is too long.')
    elif role == 'student' and '@' not in ident and not STUDENT_ID.fullmatch(ident):
        errors.append('Student ID must look like 2024-0018.')
    elif (role != 'student' or '@' in ident) and not EMAIL.fullmatch(ident):
        errors.append('Enter a valid school email address.')
    if not pw:
        errors.append('Enter your password.')
    return {'role': role, 'identifier': ident, 'password': pw}, errors


def parse_date(raw):
    """Strict YYYY-MM-DD -> date, or None if invalid."""
    try:
        return datetime.strptime(raw or '', '%Y-%m-%d').date()
    except ValueError:
        return None


def validate_marks(form, student_ids):
    """Returns ({student_id: status}, [student_ids still missing a valid status])."""
    marks = {i: form.get(f'status_{i}') for i in student_ids if form.get(f'status_{i}') in STATUSES}
    return marks, [i for i in student_ids if i not in marks]


REASON_MAX = 500

def validate_reason(form):
    """'Reason for change' on the edit screen. Returns (clean_reason, [error messages])."""
    reason = (form.get('reason') or '').strip()
    if not reason:
        return reason, ['Enter a reason for the change.']
    if len(reason) > REASON_MAX:
        return reason, [f'Reason must be {REASON_MAX} characters or fewer.']
    return reason, []

SECTION_FIELDS = {'code': ('Section code', 30), 'subject': ('Subject', 120), 'room': ('Room', 80), 'schedule': ('Schedule', 150)}

def validate_section(form, instructor_ids):
    """Returns (clean_data, [error messages]). Every problem is reported at once."""
    data = {k: ' '.join((form.get(k) or '').split()) for k in SECTION_FIELDS}      # trim + collapse spaces
    errors = []
    for key, (label, limit) in SECTION_FIELDS.items():
        if not data[key]:
            errors.append(f'{label} is required.')
        elif len(data[key]) > limit:
            errors.append(f'{label} must be {limit} characters or fewer.')
    data['instructor_id'] = form.get('instructor_id', type=int)
    if data['instructor_id'] not in instructor_ids:
        errors.append('Choose an assigned instructor from the list.')
    return data, errors


NAME_MAX = 100
EMAIL_MAX = 120

def validate_student(form, section_ids):
    """Create/edit student form. Returns (clean_data, [error messages]); uniqueness is checked by the service."""
    data = {
        'student_number': (form.get('student_number') or '').strip(),
        'name': ' '.join((form.get('name') or '').split()),
        'email': (form.get('email') or '').strip().lower(),
        'section_id': form.get('section_id', type=int),
        'active': 1 if form.get('active') == '1' else 0,
    }
    errors = []
    if not STUDENT_ID.fullmatch(data['student_number']):
        errors.append('Student ID must look like 2024-0018.')
    if not data['name']:
        errors.append('Full name is required.')
    elif len(data['name']) > NAME_MAX:
        errors.append(f'Full name must be {NAME_MAX} characters or fewer.')
    if not data['email']:
        errors.append('School email is required.')
    elif len(data['email']) > EMAIL_MAX or not EMAIL.fullmatch(data['email']):
        errors.append('Enter a valid school email address.')
    if data['section_id'] not in section_ids:
        errors.append('Choose a class section from the list.')
    return data, errors

import hmac
import secrets
from flask import abort, request, session

FIELD = 'csrf_token'
SAFE_METHODS = {'GET', 'HEAD', 'OPTIONS', 'TRACE'}


def token():
    """The signed-in visitor's CSRF token, created on first use.

    The token lives only in the Flask session, so a form posted from another site
    cannot know it.
    """
    if FIELD not in session:
        session[FIELD] = secrets.token_hex(32)
    return session[FIELD]


def init_app(app):
    @app.before_request
    def protect_from_csrf():
        if request.method in SAFE_METHODS:
            return
        sent = request.form.get(FIELD, '')
        expected = session.get(FIELD, '')
        if not expected or not hmac.compare_digest(sent, expected):
            abort(400, description='Your form session expired. Reload the page and try again.')

    app.context_processor(lambda: {'csrf_token': token})

from functools import wraps
from flask import Blueprint, abort, redirect, render_template, request, session, url_for
from werkzeug.security import check_password_hash
from db import get_db
from validators import ROLES, validate_login

bp = Blueprint('auth', __name__)
HOME = {'student': 'student.home', 'instructor': 'instructor.dashboard', 'administrator': 'admin.students'}

def login_required(f):
    @wraps(f)
    def wrapper(*a, **k):
        if 'user_id' not in session:
            return redirect(url_for('auth.login'))
        return f(*a, **k)
    return wrapper

def role_required(*roles):
    def deco(f):
        @wraps(f)
        def wrapper(*a, **k):
            if 'user_id' not in session:
                return redirect(url_for('auth.login'))
            if session['role'] not in roles:
                abort(403)
            return f(*a, **k)
        return wrapper
    return deco

def find_user(role, ident):
    """Students sign in with student ID or email; staff with email. Inactive students are blocked."""
    return get_db().execute(
        """SELECT u.*, s.id AS student_id FROM users u
           LEFT JOIN students s ON s.user_id = u.id
           WHERE u.role = ? AND (u.email = ? OR s.student_number = ?)
             AND COALESCE(s.active, 1) = 1""", (role, ident, ident)).fetchone()

@bp.route('/', methods=['GET', 'POST'])
def login():
    if request.method == 'GET' and 'role' in session:
        return redirect(url_for(HOME[session['role']]))
    role, ident, error = 'student', '', None
    if request.method == 'POST':
        data, errors = validate_login(request.form)
        role = data['role'] if data['role'] in ROLES else 'student'
        ident = data['identifier']
        if errors:
            error = errors[0]
        else:
            user = find_user(role, ident)
            if user and check_password_hash(user['password_hash'], data['password']):
                session.clear()
                session.update(user_id=user['id'], role=user['role'], name=user['name'],
                               student_id=user['student_id'])
                return redirect(url_for(HOME[role]))
            error = 'Check your ID or email and password, then try again.'
    return render_template('index.html', role=role, ident=ident, error=error)

@bp.route('/logout')
def logout():
    session.clear()
    return redirect(url_for('auth.login'))

#app.py
import os
from pathlib import Path
from flask import Flask, render_template, session, url_for
import auth, csrf, db
from routes import admin, instructor, records, reports, student

BACKEND = Path(__file__).resolve().parent
FRONTEND = BACKEND.parent / 'frontend'

ERROR_TITLES = {400: 'Request could not be completed', 403: 'Access denied',
                404: 'Page not found', 500: 'Something went wrong'}
ERROR_MESSAGES = {
    403: 'You do not have permission to open this page with your account.',
    404: 'The page you are looking for does not exist or may have been moved.',
    500: 'The server ran into a problem. Please try again in a moment.',
}


def _error_page(error):
    code = getattr(error, 'code', 500) or 500
    role = session.get('role')
    signed_in = role in auth.HOME
    home = url_for(auth.HOME[role]) if signed_in else url_for('auth.login')
    message = error.description if code == 400 and error.description else ERROR_MESSAGES.get(code, ERROR_MESSAGES[500])
    return render_template('error.html', code=code, title=ERROR_TITLES.get(code, ERROR_TITLES[500]),
                           message=message, home=home, signed_in=signed_in), code


def create_app():
    app = Flask(__name__, template_folder=FRONTEND / 'templates',
                static_folder=FRONTEND / 'asset', static_url_path='/asset')
    app.config.update(SECRET_KEY=os.environ.get('SECRET_KEY', 'dev-only-change-me'),
                      DATABASE=str(BACKEND / 'rollcall.db'))
    db.init_app(app)
    csrf.init_app(app)
    app.register_blueprint(auth.bp)
    app.register_blueprint(instructor.bp)
    app.register_blueprint(records.bp)
    app.register_blueprint(reports.bp)
    app.register_blueprint(admin.bp)
    app.register_blueprint(student.bp)
    for code in (400, 403, 404, 500):
        app.register_error_handler(code, _error_page)
    return app


if __name__ == '__main__':
    create_app().run(debug=True)

from db import get_db

PER_PAGE = 10

# Rate = (present + late) / (present + late + absent) x 100. Excused is excluded; NULL when nothing countable.
_SELECT = """
SELECT cs.id, cs.session_date, sec.code, sec.subject,
       COALESCE(SUM(a.status = 'present'), 0) AS present,
       COALESCE(SUM(a.status = 'late'), 0)    AS late,
       COALESCE(SUM(a.status = 'absent'), 0)  AS absent,
       COALESCE(SUM(a.status = 'excused'), 0) AS excused,
       CASE WHEN COALESCE(SUM(a.status IN ('present','late','absent')), 0) = 0 THEN NULL
            ELSE ROUND(100.0 * SUM(a.status IN ('present','late'))
                       / SUM(a.status IN ('present','late','absent')), 1) END AS rate
FROM class_sessions cs
JOIN sections sec ON sec.id = cs.section_id
LEFT JOIN attendance a ON a.session_id = cs.id
"""


def _like(text):
    """Escape LIKE wildcards so a search for '%' or '_' is taken literally."""
    return '%' + text.replace('\\', '\\\\').replace('%', '\\%').replace('_', '\\_') + '%'


def _where(user_id, section_id, start, end, q):
    where, args = ['sec.instructor_id = ?'], [user_id]       # instructors only ever see their own sections
    if section_id:
        where.append('cs.section_id = ?'); args.append(section_id)
    if start:
        where.append('cs.session_date >= ?'); args.append(start.isoformat())
    if end:
        where.append('cs.session_date <= ?'); args.append(end.isoformat())
    if q:
        like = _like(q)
        where.append("""(sec.code LIKE ? ESCAPE '\\' OR sec.subject LIKE ? ESCAPE '\\' OR EXISTS (
                          SELECT 1 FROM attendance a2 JOIN students s ON s.id = a2.student_id
                          WHERE a2.session_id = cs.id
                            AND (s.name LIKE ? ESCAPE '\\' OR s.student_number LIKE ? ESCAPE '\\')))""")
        args += [like] * 4
    return ' AND '.join(where), args


def count_sessions(user_id, section_id=None, start=None, end=None, q=''):
    where, args = _where(user_id, section_id, start, end, q)
    return get_db().execute('SELECT COUNT(*) FROM class_sessions cs JOIN sections sec ON sec.id = cs.section_id '
                            f'WHERE {where}', args).fetchone()[0]


def list_sessions(user_id, section_id=None, start=None, end=None, q='', page=1):
    where, args = _where(user_id, section_id, start, end, q)
    return get_db().execute(
        f'{_SELECT} WHERE {where} GROUP BY cs.id ORDER BY cs.session_date DESC, sec.code LIMIT ? OFFSET ?',
        args + [PER_PAGE, (page - 1) * PER_PAGE]).fetchall()


def get_session(session_id, user_id):
    """The saved session with its section, only if the section belongs to this instructor."""
    return get_db().execute(
        'SELECT cs.id, cs.session_date, sec.id AS section_id, sec.code, sec.subject FROM class_sessions cs '
        'JOIN sections sec ON sec.id = cs.section_id WHERE cs.id = ? AND sec.instructor_id = ?',
        (session_id, user_id)).fetchone()


def apply_edit(session_id, user_id, changes, reason):
    """One transaction: every changed status and the edit_log row are saved together or not at all."""

    with con:
        con.executemany('UPDATE attendance SET status = ? WHERE session_id = ? AND student_id = ?',
                        [(status, session_id, student) for student, status in changes.items()])
        con.execute('INSERT INTO edit_log(session_id, user_id, reason) VALUES (?,?,?)',
                    (session_id, user_id, reason))

from db import get_db

CHART_SESSIONS = 10      # the bar chart shows the most recent sessions only


def rate(present, late, absent):
    """(Present + Late) / (Present + Late + Absent) x 100. Excused is excluded. None if nothing countable."""
    counted = present + late + absent
    return round(100 * (present + late) / counted, 1) if counted else None


def report_sections(user_id=None):
    """Sections a report may cover: the instructor's own, or every section when user_id is None (admin)."""
    sql, args = 'SELECT * FROM sections', []
    if user_id is not None:
        sql, args = sql + ' WHERE instructor_id = ?', [user_id]
    return get_db().execute(sql + ' ORDER BY code', args).fetchall()


def _where(user_id, section_id, start, end):
    where, args = ['1 = 1'], []
    if user_id is not None:                      # instructors only ever see their own sections
        where.append('sec.instructor_id = ?'); args.append(user_id)
    if section_id:
        where.append('cs.section_id = ?'); args.append(section_id)
    if start:
        where.append('cs.session_date >= ?'); args.append(start.isoformat())
    if end:
        where.append('cs.session_date <= ?'); args.append(end.isoformat())
    return ' AND '.join(where), args


_COUNTS = """COALESCE(SUM(a.status = 'present'), 0) AS present, COALESCE(SUM(a.status = 'late'), 0) AS late,
             COALESCE(SUM(a.status = 'absent'), 0) AS absent, COALESCE(SUM(a.status = 'excused'), 0) AS excused"""
_FROM = """FROM class_sessions cs JOIN sections sec ON sec.id = cs.section_id
           LEFT JOIN attendance a ON a.session_id = cs.id"""


def build_report(user_id=None, section_id=None, start=None, end=None):
    """One report for both roles. Returns totals, per-session rows (chart) and per-student rows (table)."""
    con = get_db()
    where, args = _where(user_id, section_id, start, end)

    t = con.execute(f"SELECT COUNT(DISTINCT cs.id) AS sessions, COUNT(DISTINCT a.student_id) AS students, {_COUNTS} "
                    f"{_FROM} WHERE {where}", args).fetchone()
    totals = dict(t, rate=rate(t['present'], t['late'], t['absent']))

    sessions = [dict(r, rate=rate(r['present'], r['late'], r['absent'])) for r in con.execute(
        f"SELECT cs.id, cs.session_date, sec.code, {_COUNTS} {_FROM} WHERE {where} "
        'GROUP BY cs.id ORDER BY cs.session_date DESC, sec.code DESC LIMIT ?', args + [CHART_SESSIONS])][::-1]

    students = [dict(r, rate=rate(r['present'], r['late'], r['absent'])) for r in con.execute(
        f"SELECT s.id, s.student_number, s.name, s.active, sec.code, {_COUNTS} {_FROM} "
        f"JOIN students s ON s.id = a.student_id WHERE {where} GROUP BY s.id ORDER BY s.name", args)]
    students.sort(key=lambda r: (r['rate'] is None, r['rate'] if r['rate'] is not None else 0))  # lowest rate first
    return {'totals': totals, 'sessions': sessions, 'students': students}

                    (data['student_number'], data['name'], data['section_id'], data['active'], student_id))

@bp.route('/attendance', methods=['GET', 'POST'])
@role_required('instructor')
def take_attendance():
    uid = session['user_id']
    src = request.form if request.method == 'POST' else request.args
    section_id, day = src.get('section_id', type=int), src.get('date') or date.today().isoformat()
    ctx = dict(sections=svc.instructor_sections(uid), section_id=section_id, day=day, today=date.today().isoformat(),
               section=None, existing=None, rows=[], missing=set(), notice=None)

    section = svc.get_section(section_id, uid) if section_id else None
    d = parse_date(day)
    if section_id and not section:
        ctx['notice'] = ('error', 'Choose one of your assigned sections.')
    elif section and not d:
        ctx['notice'] = ('error', 'Enter a valid session date.')
    elif section:
        ctx['section'] = section
        existing = svc.find_session(section['id'], day)
        if existing:                                   # already saved: show it read-only
            ctx.update(existing=existing, rows=svc.session_rows(existing['id']))
            if request.method == 'POST':
                ctx['notice'] = ('error', 'Attendance for this section and date is already saved. Use Edit record to change it.')
            elif request.args.get('saved'):
                ctx['notice'] = ('success', f'Attendance saved successfully · {section["code"]} · {d:%B %d, %Y} · {len(ctx["rows"])} students')
        else:
            rows = svc.roster(section['id'])
            ctx['rows'] = rows
            if not rows:
                ctx['notice'] = ('warning', 'This section has no active students.')
            elif request.method == 'POST':
                marks, missing = validate_marks(request.form, [r['id'] for r in rows])
                for r in rows:                         # keep the instructor's selections
                    r['status'] = marks.get(r['id'])
                if d > date.today():
                    ctx['notice'] = ('error', 'Session date cannot be in the future.')
                elif missing:
                    n = len(missing)
                    ctx.update(missing=set(missing), notice=('warning',
                               f'{n} student{"s still need" if n > 1 else " still needs"} a status. '
                               'Mark every student before saving.'))
                else:
                    try:
                        svc.save_session(section['id'], day, uid, marks)
                    except sqlite3.IntegrityError:
                        ctx['notice'] = ('error', 'Could not save: a record for this section and date already exists.')
                    except sqlite3.Error:
                        ctx['notice'] = ('error', 'Could not save attendance because of a database error. Please try again.')
                    else:
                        return redirect(url_for('instructor.take_attendance', section_id=section['id'], date=day, saved=1))
    return render_template('attendance.html', **ctx)

        notice = ('success', 'Attendance changes saved successfully.')

    original = {r['id']: r['status'] for r in rows}
    reason, notice, missing = '', None, set()

    if request.method == 'POST':
        marks, missing_ids = validate_marks(request.form, list(original))
        reason, errors = validate_reason(request.form)
        for r in rows:                  # keep the instructor's selections when the form re-renders
            r['status'] = marks.get(r['id'])
        if missing_ids:
            n = len(missing_ids)
            missing = set(missing_ids)
            notice = ('warning', f'{n} student{"s still need" if n > 1 else " still needs"} a status. '
                                 'Mark every student before saving.')
        elif errors:
            notice = ('error', errors[0])
        else:
            changes = {i: s for i, s in marks.items() if s != original[i]}
            if not changes:
                notice = ('warning', 'No statuses were changed. Change at least one status before saving.')
            else:
                try:
                    svc.apply_edit(session_id, uid, changes, reason)
                except sqlite3.Error:
                    notice = ('error', 'Could not save the changes because of a database error. Please try again.')
                else:
                    return redirect(url_for('records.index', updated=1))
    return render_template('edit-record.html', rec=rec, rows=rows, missing=missing, reason=reason, notice=notice)

bp = Blueprint('reports', __name__)


def _show(user_id, admin):
    """Shared by both roles. user_id=None means 'all sections' (administrator)."""
    sections = svc.report_sections(user_id)
    section_id = request.args.get('section_id', type=int)
    start_raw, end_raw = request.args.get('start_date', '').strip(), request.args.get('end_date', '').strip()
    start, end = parse_date(start_raw), parse_date(end_raw)

    notice = None
    if section_id and section_id not in {s['id'] for s in sections}:
        notice = ('error', 'Choose one of the listed sections.')
    elif start_raw and not start:
        notice = ('error', 'Enter a valid From date.')
    elif end_raw and not end:
        notice = ('error', 'Enter a valid To date.')
    elif start and end and start > end:
        notice = ('error', 'The From date must be on or before the To date.')

    report = None if notice else svc.build_report(user_id, section_id, start, end)
    return render_template('reports.html', admin=admin, sections=sections, section_id=section_id,
                           start=start_raw, end=end_raw, notice=notice, report=report,
                           action=request.path)

def _duplicate_errors(data, student=None):
    """Friendly duplicate messages. student is the existing row when editing."""
    errors = []
    if stu_svc.number_taken(data['student_number'], student['id'] if student else None):
        errors.append('That student ID is already in use.')
    if stu_svc.email_taken(data['email'], student['user_id'] if student else None):
        errors.append('That school email is already registered.')
    return errors


def _save_error(action):
    return f'Could not {action} because of a database error. Please try again.'

@bp.route('/students/add', methods=['GET', 'POST'])
@role_required('administrator')
def add_student():
    sections = sec_svc.list_sections()
    data = {'student_number': '', 'name': '', 'email': '', 'section_id': None, 'active': 1}
    errors = []
    if request.method == 'POST':
        data, errors = validate_student(request.form, {s['id'] for s in sections})
        errors += _duplicate_errors(data)
        if not errors:
            try:
                stu_svc.save(data)
            except sqlite3.IntegrityError:
                errors.append('That student ID or school email is already in use.')
            except sqlite3.Error:
                errors.append(_save_error('save the student'))
            else:
                return redirect(url_for('admin.students', saved=1))
    return render_template('student-form.html', student=None, data=data, errors=errors, sections=sections)

<script>
  const rows = [...document.querySelectorAll('tbody tr[data-row]')];
  const $ = id => document.getElementById(id);
  function refresh() {
    const c = {present: 0, late: 0, absent: 0, excused: 0};
    rows.forEach(r => { const s = r.querySelector('input:checked'); if (s) c[s.value]++; });
    for (const k in c) $('n-' + k).textContent = c[k];
    $('marked').textContent = Object.values(c).reduce((a, b) => a + b, 0) + ' / ' + rows.length + ' marked';
  }
  document.addEventListener('change', e => {
    if (e.target.matches('.roster input[type=radio]')) { e.target.closest('tr').classList.remove('missing-row'); refresh(); applyFilters(); }
  });
  let mode = 'all';                                  // all | marked | unmarked
  function applyFilters() {
    const q = $('roster-search').value.trim().toLowerCase();
    let shown = 0;
    rows.forEach(r => {
      const marked = !!r.querySelector('input:checked');
      const okMode = mode === 'all' || (mode === 'marked') === marked;
      r.hidden = !(okMode && r.dataset.row.includes(q));
      shown += !r.hidden;
    });
    $('no-match').hidden = shown > 0;
  }
  $('roster-search').addEventListener('input', applyFilters);
  [['view-marked', 'marked'], ['review-unmarked', 'unmarked']].forEach(([id, m]) => {
    $(id).addEventListener('click', () => {
      mode = mode === m ? 'all' : m;                 // click again to show everyone
      ['view-marked', 'review-unmarked'].forEach(b => {
        const on = (b === 'view-marked' ? 'marked' : 'unmarked') === mode;
        $(b).setAttribute('aria-pressed', on); $(b).classList.toggle('btn-soft', on); $(b).classList.toggle('btn-secondary', !on);
      });
      applyFilters();
    });
  });
  refresh();
</script>

# Rollcall — Student Attendance Monitoring System

Software Engineering 2 · Laboratory Activity 6 · Task 3: Software Implementation

Team: Balagao, Insigne, Macalindong  
Submitted to: Gerald Jane Salacsacan · October 2026

Rollcall is a working web system for recording and reviewing class attendance. It has three roles
(student, instructor, administrator), a SQLite database, server-side validation, and a Jinja front end
based on the original HTML/CSS prototype.

## Requirements

- Python 3.10 or newer
- Flask 3 (`backend/requirements.txt`)

## Setup and run (Windows, VS Code)

The virtual environment is already in `venv/` at the project root.

```powershell
cd backend
..\venv\Scripts\python.exe -m pip install -r requirements.txt
..\venv\Scripts\python.exe seed.py      # creates rollcall.db with demo data
..\venv\Scripts\python.exe app.py
```

Open <http://127.0.0.1:5000>.

To start over with clean demo data, stop the server and run `python seed.py` again.

## Demo accounts

| Role          | Sign in                                                    | Password         |
| ------------- | ---------------------------------------------------------- | ---------------- |
| Administrator | `admin@school.edu`                                         | `Admin@123`      |
| Instructor    | `instructor@school.edu`                                    | `Instructor@123` |
| Student       | `2024-0018` (Alex Reyes) or `alex.reyes@student.rollcall.local` | `Student@123`    |

Every seeded student uses `Student@123`. A student created from the admin screen also starts with
`Student@123` (stored hashed).

Seed contents: 4 sections (BSIT 2A, 2B, 2C, 3A), 32 students each, 8 saved sessions per section
(2026-10-01 to 2026-10-22), and 1,024 attendance rows.

## What the system does

- **Sign in by role** with a student ID or school email and password. Inactive students cannot sign in.
- **Take attendance** for the instructor's own sections: choose section and date, load the roster,
  mark present/late/absent/excused, see live counts, search the roster, and save. Incomplete rosters
  and future dates are rejected with clear messages; a saved section/date cannot be saved twice.
- **Attendance records**: filter by section, date range, or search; paginate; open a record and edit
  statuses with a required reason. Every edit is written to `edit_log`.
- **Reports** for instructors and administrators: overall rate, session/student totals, a per-session
  bar chart, and a per-student breakdown. Excused marks are excluded from the rate.
- **Admin students**: search by name/ID, filter by section, add, edit, and soft delete by setting
  Enrollment status to Inactive. Creating a student also creates the linked user account.
- **Admin sections**: search, add, and edit sections; unique code and a valid instructor are required;
  each card shows the active student count.
- **Student self-view**: own rate, present/late/absent counts, and session history with an optional
  subject filter. The student is resolved from the login session, never from the URL.
- **Instructor overview**: assigned classes, enrolled students, overall rate, sessions saved today,
  and a chart of the last 5 sessions.

Rate = (Present + Late) ÷ (Present + Late + Absent) × 100. Excused is excluded; the rate shows “—” when
there is nothing countable.

## Validation, errors, and security

- Inputs are validated on the server (role, login identifier, student ID format `YYYY-NNNN`, names,
  email, section and instructor references, dates, statuses, required edit reasons, max lengths).
- Duplicate student IDs, emails, and section codes produce friendly messages.
- Passwords are hashed with Werkzeug.
- Writers only see and edit their own sections; role checks return a custom 403 page.
- Custom 400/403/404/500 pages replace Flask's default pages.
- CSRF tokens are required on every POST form, including sign-in.
- The sign-in and error pages use `frontend/asset/landing_bg.jpg` as a background.

## Project layout

```
backend/
  app.py, auth.py, csrf.py, db.py, schema.sql, seed.py, validators.py
  requirements.txt, rollcall.db (created by seed.py)
  routes/       auth (in auth.py), instructor, records, reports, admin, student
  services/     attendance, records, reports, students, sections, student, dashboard
frontend/
  asset/        styles.css, landing_bg.jpg
  templates/    base, index (sign-in), error, attendance, edit-record, records,
                dashboard, reports, students, student-form, sections, section-form, student
```

## Notes

- The dev server uses `debug=True` for the demo; set a real `SECRET_KEY` environment variable for
  anything beyond the classroom demo. With `debug=True`, Flask shows its debugger for unexpected
  exceptions; the custom 500 page is used when debug is off.
- The unused prototype files from Task 2C are kept under `frontend/templates/` for reference
  (`design-notes.html`, `attendance-*.html`, `login-error.html`, `record-updated.html`,
  `admin-reports.html`). They are not served by the application.

   - Stats read **128 students shown**, **4 class sections**, **128 active enrollments**.
2. Search `Alex` and filter **BSIT 2A**. Show the matching row and the Active chip.
3. Click **Add student** and enter:
   - Student ID `2024-9001`, name `Isabel Santos`, email `isabel.santos@school.edu`, section `BSIT 2A`.
   - Submit. Point out the success notice and the new row.
4. Add a second student with the same ID but a different email.
   - Show the friendly message: “That student ID is already in use.” Cancel.
5. Edit Isabel, change **Enrollment status** to **Inactive**, save.
   - The chip changes to Inactive. Mention that this is the soft delete: history is kept and the
     student can no longer sign in.
6. Open **Class sections**. Point out the student count on each card. Open **Add section**, enter
   code `BSIT 4A`, then try the same code in lowercase — it is rejected. Cancel (or edit an existing
   section’s room and save).

What to say: “The admin manages the master data; validation and duplicate checks happen on the
server, not only in the browser.”

## 2. Instructor — attendance, records, reports (~3 minutes)

1. Sign out, then sign in as `instructor@school.edu` / `Instructor@123`.
   - The **Overview** shows 4 assigned classes, 128 enrolled students, an overall rate, sessions
     saved today, and a chart of the last 5 sessions.
2. Click **Take attendance**. Choose **BSIT 2A · Database Management**, date `2026-10-23`, and load
   the list.
3. Press **Save attendance** without marking anyone.
   - Show the warning that every student needs a status; nothing is saved.
4. Click **Mark remaining as present**, then save.
   - Show the success notice with the section, date, and student count.
5. Open **Attendance records**. The new 2026-10-23 session is at the top with counts and rate.
   - Instructors only see their own sections; this is enforced in SQL.
6. Open **Edit record** on that session. Change one student to **Absent**, type a reason, and save.
   - Show the success notice and the updated counts.
   - Try saving with no reason, or with no changed status, to show the validation messages.
7. Open **Reports**. Show the rate, sessions, student breakdown, status counts, and bar chart.
   - Filter by section or date range and generate again.

What to say: “A whole session is saved in one transaction; an edit saves the status changes and the
edit-log row together.”

## 3. Student — own attendance only (~1 minute)

1. Sign out, then sign in as `2024-0006` / `Student@123` (Hana Castro).
   - Show 8 sessions, 5 present, 1 late, 2 absent, rate **75.0%**.
2. Use the **Subject** filter, then clear it with **All subjects**.
3. In the address bar, add `?student_id=1` to `/student`.
   - It still shows Hana. Say: “The student ID comes from the signed-in session, never from the URL.”
4. Try `/students` while signed in as the student.
   - Show the custom **403 Access denied** page.

What to say: “Each student can only ever read their own rows.”

## 4. Error handling and security (~1 minute, optional)

- Visit `/does-not-exist` — custom **404** page with a link back to the dashboard.
- While signed in as student, visit `/instructor` — custom **403** page.
- Optional CSRF proof from PowerShell (server running, no session cookie):

  ```powershell
  Invoke-WebRequest -Method Post -Uri http://127.0.0.1:5000/ `
    -Body @{ role = 'student'; identifier = '2024-0018'; password = 'Student@123' }
  ```

  The response is the custom **400** page because the POST has no CSRF token. The browser forms all
  send the token, so normal use is unaffected.

Finish by signing out.

- Three working features: attendance marking, records/editing, reports, plus admin management and the
  student view.
- Appropriate UI: shared sidebar/topbar, cards, tables, charts, responsive CSS.
- Input validation: formats, required fields, dates, statuses, duplicate checks, ownership checks.
- Correct data handling: parameterized SQL, transactions, the agreed rate formula, soft delete.
- Error messages: friendly inline notices, incomplete-roster warnings, duplicate messages, custom
  error pages.
- Organized code: blueprints, services, validators, schema, and reusable templates.
